export default function App(): React.ReactElement {
    const [currentPath, setCurrentPath] = useState<string>("/");

    const handleNavigate = (path: string): void => {
        setCurrentPath(path);
    };

    const renderContent = () => {
        switch (currentPath) {
            case "/":
                return <HomePageView onNavigate={handleNavigate} />;
            case "/conserto-celular-tablet":
                return (
                    <GenericPageView
                        title="Conserto Especializado de Celular & Tablet"
                        desc="Troca de telas originais, substituição de bateria com garantia de 90 dias, desoxidação e reparo em placa."
                        onBack={() => handleNavigate("/")}
                    />
                );
            case "/manutencao-computador-notebook":
                return (
                    <GenericPageView
                        title="Manutenção de Computador & Notebook"
                        desc="Limpeza preventiva térmica, upgrades de memória RAM e SSD NVMe, reparo eletrônico em placas mãe."
                        onBack={() => handleNavigate("/")}
                    />
                );
            case "/atendimento-domiciliar-empresas":
                return (
                    <GenericPageView
                        title="Atendimento Domiciliar & Empresas (B2B)"
                        desc="Gestão e manutenção técnica corporativa com contrato SLA, suporte presencial e atendimento leva e traz seguro."
                        onBack={() => handleNavigate("/")}
                    />
                );
            case "/sobre-nos":
                return (
                    <GenericPageView
                        title="Sobre a Infinic Assistência Técnica"
                        desc="Conheça nossa bancada com instrumentação de ponta, microscópios de precisão e equipe qualificada."
                        onBack={() => handleNavigate("/")}
                    />
                );
            case "/orcamento":
                return (
                    <GenericPageView
                        title="Solicitação de Orçamento Online"
                        desc="Envie as especificações do seu aparelho e receba a prévia de custos e prazos de atendimento."
                        onBack={() => handleNavigate("/")}

            case "/status-os":

                        desc="Digite o número da sua OS para acompanhar o diagnóstico, reparo e liberação para retirada."
                        onBack={() => handleNavigate("/")}
                    />
                );
            default:
                return <HomePageView onNavigate={handleNavigate} />;
        }
    };

    return (
        <SiteLayout activePath={currentPath} onNavigate={handleNavigate}>
            {renderContent()}
        </SiteLayout>
